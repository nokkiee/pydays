@solution
import csv


def load_people(filename):
    result = []
    with open(filename, encoding='utf-8', newline='') as f:
        for row in csv.DictReader(f):
            for key in ('age', 'height_cm', 'weight_kg'):
                row[key] = int(row[key])
            result.append(row)
    return result


people = load_people('people.csv')
average_age = sum(p['age'] for p in people) / len(people)
print(average_age)

@solution
import csv

with open('people.csv', encoding='utf-8', newline='') as f:
    people = [{**r, 'age': int(r['age']), 'height_cm': int(r['height_cm']), 'weight_kg': int(r['weight_kg'])} for r in csv.DictReader(f)]


def average_by(rows, key, value):
    groups = {}
    for row in rows:
        groups.setdefault(row[key], []).append(row[value])
    return {k: sum(v) / len(v) for k, v in groups.items()}


print(average_by(people, 'country', 'age'))
@end
