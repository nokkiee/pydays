en: ## REST and CRUD
Yesterday you used APIs; today you build one. Most web APIs follow the **REST** style: the data is organised as **resources** with their own URLs, such as `/students` for all students and `/students/3` for one of them, and the HTTP method says what to do with it. The four basic operations on stored data are called **CRUD**:

| Operation | Method | URL | Answer |
|---|---|---|---|
| Create | `POST` | `/students` | `201` and the new student |
| Read all | `GET` | `/students` | `200` and a list |
| Read one | `GET` | `/students/3` | `200`, or `404` if there is none |
| Update | `PUT` | `/students/3` | `200` and the changed student |
| Delete | `DELETE` | `/students/3` | `204` with no body |

Requests and answers carry JSON. A request with missing or wrong data gets `400` and a short error message.

@code norun needs=flask
from flask import Flask, jsonify, request

app = Flask(__name__)
students = {1: {'id': 1, 'name': 'Asabeneh', 'country': 'Finland'}}
next_id = 2


@app.route('/students', methods=['GET'])
def list_students():
    return jsonify(list(students.values()))


@app.route('/students/<int:sid>', methods=['GET'])
def get_student(sid):
    if sid not in students:
        return jsonify({'error': 'not found'}), 404
    return jsonify(students[sid])


@app.route('/students', methods=['POST'])
def create_student():
    global next_id
    data = request.get_json() or {}
    if 'name' not in data:
        return jsonify({'error': 'name is required'}), 400
    student = {'id': next_id, **data}
    students[next_id] = student
    next_id += 1
    return jsonify(student), 201


@app.route('/students/<int:sid>', methods=['DELETE'])
def delete_student(sid):
    students.pop(sid, None)
    return '', 204


if __name__ == '__main__':
    app.run(debug=True)
@end

@solution
students = {
    1: {'id': 1, 'name': 'Asabeneh', 'country': 'Finland'},
    2: {'id': 2, 'name': 'Anna', 'country': 'Poland'},
}
NOT_FOUND = (404, {'error': 'not found'})


def handle(method, path, body=None):
    parts = path.strip('/').split('/')
    if parts[0] != 'students' or len(parts) > 2:
        return NOT_FOUND
    if len(parts) == 1:
        if method == 'GET':
            return 200, list(students.values())
        if method == 'POST':
            body = body or {}
            if not body.get('name'):
                return 400, {'error': 'name is required'}
            new_id = max(students, default=0) + 1
            student = {**body, 'id': new_id}
            students[new_id] = student
            return 201, student
        return NOT_FOUND
    if method == 'GET' and parts[1].isdigit():
        sid = int(parts[1])
        if sid in students:
            return 200, students[sid]
    return NOT_FOUND


print(handle('POST', '/students', {'name': 'Lukas', 'country': 'Germany'}))
print(handle('POST', '/students', {'country': 'Spain'}))

students = {
    1: {'id': 1, 'name': 'Asabeneh', 'country': 'Finland'},
    2: {'id': 2, 'name': 'Anna', 'country': 'Poland'},
}
NOT_FOUND = (404, {'error': 'not found'})


def handle(method, path, body=None):
    parts = path.strip('/').split('/')
    if parts[0] != 'students' or len(parts) > 2:
        return NOT_FOUND
    if len(parts) == 1:
        if method == 'GET':
            return 200, list(students.values())
        if method == 'POST':
            body = body or {}
            if not body.get('name'):
                return 400, {'error': 'name is required'}
            new_id = max(students, default=0) + 1
            student = {**body, 'id': new_id}
            students[new_id] = student
            return 201, student
        return NOT_FOUND
    if not parts[1].isdigit() or int(parts[1]) not in students:
        return NOT_FOUND
    sid = int(parts[1])
    if method == 'GET':
        return 200, students[sid]
    if method == 'PUT':
        students[sid].update(body or {})
        students[sid]['id'] = sid
        return 200, students[sid]
    if method == 'DELETE':
        del students[sid]
        return 204, None
    return NOT_FOUND


print(handle('PUT', '/students/2', {'country': 'Germany'}))
print(handle('DELETE', '/students/1'))
print(handle('GET', '/students'))
