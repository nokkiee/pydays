fr: Utilise une boucle `while True:` avec `break` sur une ligne vide. `re.findall(r"[a-z']+", text.lower())` trouve les mots ; trie avec `key=lambda p: (-p[1], p[0])`.
@starter
import re

lines = []
@end
@solution
import re

lines = []
while True:
    line = input()
    if not line:
        break
    lines.append(line)

counts = {}
for word in re.findall(r"[a-z']+", ' '.join(lines).lower()):
    counts[word] = counts.get(word, 0) + 1

for word, n in sorted(counts.items(), key=lambda p: (-p[1], p[0]))[:3]:
    print(f'{word}: {n}')
@end

@starter
import json

@end
@solution
import json


class TodoList:
    def __init__(self):
        self.tasks = []

    def add(self, title):
        self.tasks.append({'title': title, 'done': False})

    def complete(self, title):
        for task in self.tasks:
            if task['title'] == title and not task['done']:
                task['done'] = True
                return True
        return False

    def pending(self):
        return [t['title'] for t in self.tasks if not t['done']]

    def save(self, filename):
        with open(filename, 'w', encoding='utf-8') as f:
            json.dump(self.tasks, f, ensure_ascii=False, indent=2)

    @classmethod
    def load(cls, filename):
        todo = cls()
        with open(filename, encoding='utf-8') as f:
            todo.tasks = json.load(f)
        return todo


todo = TodoList()
todo.add('Finish day 30')
todo.add('Start a project')
todo.complete('Finish day 30')
print(todo.pending())
@end

@starter
def fizzbuzz(n):

@end
@solution
def fizzbuzz(n):
    result = []
    for i in range(1, n + 1):
        if i % 15 == 0:
            result.append('FizzBuzz')
        elif i % 3 == 0:
            result.append('Fizz')
        elif i % 5 == 0:
            result.append('Buzz')
        else:
            result.append(str(i))
    return result


for item in fizzbuzz(15):
    print(item)
@end
