en: ## What is web scraping?
**Web scraping** means collecting data from web pages with a program instead of copying it by hand. A page is written in **HTML**: nested **tags** like `<h1>`, `<p>`, `<a>` or `<table>`, which can carry **attributes** – `<a href="...">` holds the address of a link, and `class` and `id` name elements so that they can be found.

Scrape responsibly: check a site's terms and its `robots.txt`, do not send many requests in a short time, and prefer an official API when there is one (day 28).

en: ## requests and BeautifulSoup
The usual tools are the packages `requests`, which downloads a page, and `beautifulsoup4`, which parses the HTML so you can search it: `find` returns the first matching element, `find_all` all of them, and `get_text()` the text inside. Install them with `python -m pip install requests beautifulsoup4`.

from html.parser import HTMLParser

class TagPrinter(HTMLParser):
    def handle_starttag(self, tag, attrs):
        print('start:', tag, dict(attrs))

    def handle_data(self, data):
        if data.strip():
            print('text: ', data.strip())

parser = TagPrinter()

from html.parser import HTMLParser

class TitleFinder(HTMLParser):
    def __init__(self):
        super().__init__()
        self.in_title = False
        self.title = ''

    def handle_starttag(self, tag, attrs):
        if tag == 'title':
            self.in_title = True

    def handle_endtag(self, tag):
        if tag == 'title':
            self.in_title = False

    def handle_data(self, data):
        if self.in_title:
            self.title += data

with open('page.html', encoding='utf-8') as f:
    finder = TitleFinder()
    finder.feed(f.read())

from html.parser import HTMLParser


class LinkCollector(HTMLParser):
    def __init__(self):
        super().__init__()
        self.links = []

    def handle_starttag(self, tag, attrs):
        if tag == 'a':
            href = dict(attrs).get('href')
            if href:
                self.links.append(href)


collector = LinkCollector()
with open('page.html', encoding='utf-8') as f:
    collector.feed(f.read())
links = collector.links

from html.parser import HTMLParser


class HeadingCollector(HTMLParser):
    def __init__(self):
        super().__init__()
        self.headings = []
        self.inside = False

    def handle_starttag(self, tag, attrs):
        if tag == 'h2':
            self.inside = True
            self.headings.append('')

    def handle_endtag(self, tag):
        if tag == 'h2':
            self.inside = False

    def handle_data(self, data):
        if self.inside:
            self.headings[-1] += data


parser = HeadingCollector()

    parser.feed(f.read())

@solution
from html.parser import HTMLParser


class TableParser(HTMLParser):
    def __init__(self):
        super().__init__()
        self.rows = []
        self.in_cell = False

    def handle_starttag(self, tag, attrs):
        if tag == 'tr':
            self.rows.append([])
        elif tag == 'td':
            self.in_cell = True
            self.rows[-1].append('')

    def handle_endtag(self, tag):
        if tag == 'td':
            self.in_cell = False

    def handle_data(self, data):
        if self.in_cell:
            self.rows[-1][-1] += data


parser = TableParser()
with open('page.html', encoding='utf-8') as f:
    parser.feed(f.read())

books = []
for row in parser.rows:
    if row:
        title, price, stock = row
        books.append({'title': title, 'price': float(price), 'in_stock': stock == 'yes'})
cheapest = min((b for b in books if b['in_stock']), key=lambda b: b['price'])['title']
print(books)
print(cheapest)
@end
