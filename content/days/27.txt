en: ## Finding, updating and deleting
`find_one(query)` returns the first matching document and `find(query)` all of them. A **query** is a dictionary: `{'country': 'Finland'}` matches documents whose `country` is Finland. Operators start with `$`: `{'age': {'$gt': 30}}` means age greater than 30; others are `$lt`, `$gte`, `$lte`, `$ne` and `$in`. `update_one(query, {'$set': {...}})` changes fields, and `delete_one` and `delete_many` remove documents. Every document gets a unique `_id`.

Każdy dokument dostaje unikalne `_id`.

OPS = {
    '$gt': lambda a, b: a > b,
    '$gte': lambda a, b: a >= b,
    '$lt': lambda a, b: a < b,
    '$lte': lambda a, b: a <= b,
    '$ne': lambda a, b: a != b,
    '$in': lambda a, b: a in b,
}


def matches(doc, query):
    for field, condition in query.items():
        if isinstance(condition, dict):
            for op, arg in condition.items():
                if field not in doc:
                    if op != '$ne':
                        return False
                    continue
                if not OPS[op](doc[field], arg):
                    return False
        elif doc.get(field) != condition:
            return False
    return True


print(matches({'name': 'Sami', 'age': 25}, {'age': {'$gt': 20, '$lt': 30}}))

class Collection:
    def __init__(self):
        self.docs = []
        self.next_id = 1

    def insert_one(self, doc):
        stored = {**doc, '_id': self.next_id}
        self.next_id += 1
        self.docs.append(stored)
        return stored['_id']

    def find(self, query={}):
        return [d for d in self.docs if matches(d, query)]

    def find_one(self, query={}):
        found = self.find(query)
        return found[0] if found else None

    def count(self, query={}):
        return len(self.find(query))

def matches(doc, query):
    return all(doc.get(field) == value for field, value in query.items())


class Collection:
    def __init__(self):
        self.docs = []
        self.next_id = 1

    def insert_one(self, doc):
        stored = {**doc, '_id': self.next_id}
        self.next_id += 1
        self.docs.append(stored)
        return stored['_id']

    def find(self, query={}):
        return [d for d in self.docs if matches(d, query)]

    def update_one(self, query, update):
        found = self.find(query)
        if not found:
            return 0
        found[0].update(update['$set'])
        return 1

    def delete_many(self, query):
        before = len(self.docs)
        self.docs = [d for d in self.docs if not matches(d, query)]
        return before - len(self.docs)


students = Collection()
for name, city in [('Asabeneh', 'Helsinki'), ('David', 'London'), ('Eyob', 'London')]:
    students.insert_one({'name': name, 'city': city})
students.update_one({'name': 'Asabeneh'}, {'$set': {'city': 'Espoo'}})
print(students.delete_many({'city': 'London'}), students.docs)
